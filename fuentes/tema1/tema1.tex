\section{Sistemas termodinámicos}
% =====================================================================

La termodinámica describe sistemas de $N\sim 10^{23}$ partículas mediante unas pocas magnitudes
macroscópicas ($p$, $V$, $T$, $n$, \dots). La clave es que las medidas macroscópicas son
\emph{lentas} y \emph{gruesas} frente a las escalas atómicas: solo sobreviven al promedio
estadístico las combinaciones de coordenadas que no fluctúan, y estas definen el
\emph{macroestado}~\cite{apuntes,transp}.

\subsection{Sistema, frontera y paredes}

\begin{definicion}[Magnitud y sistema termodinámico]
Una \emph{magnitud termodinámica} es una propiedad macroscópica que describe el estado interno de un
sistema. Un \emph{sistema termodinámico} es una región del espacio descrita por magnitudes
termodinámicas, separada de sus \emph{alrededores} por una \emph{frontera} (real o conceptual, fija o
móvil). Sistema, frontera y alrededores relevantes forman el \emph{universo termodinámico}.
\end{definicion}

\begin{definicion}[Aislado, cerrado, abierto]
Un sistema es \emph{aislado} si no intercambia ni energía ni materia; \emph{cerrado} si intercambia
energía pero no materia; \emph{abierto} si intercambia ambas.
\end{definicion}

La frontera se caracteriza por las \emph{paredes}, que permiten o impiden cada tipo de interacción
(tabla~\ref{tab:paredes}). Una \emph{ligadura} elimina una interacción posible y reduce los cambios
accesibles al sistema.

\begin{definicion}[Fase, especie y componente]
Un sistema es \emph{homogéneo} (monofásico) si sus propiedades son las mismas en toda su
extensión; si consta de varias regiones homogéneas (\emph{fases}) separadas por interfaces, es
\emph{heterogéneo}. Una \emph{especie química} es una entidad identificable por análisis químico; un
\emph{componente} es una especie independiente necesaria para fijar la composición.
\end{definicion}

\subsection{Variables extensivas e intensivas}

\begin{definicion}[Extensiva e intensiva]
Una variable es \emph{extensiva} si es aditiva sobre las partes del sistema ($U$, $V$, $n$, $m$, $S$):
al unir dos copias idénticas se duplica. Es \emph{intensiva} si es local y no aditiva ($T$, $p$,
$\rho$): al duplicar el sistema no cambia.
\end{definicion}

El cociente de dos extensivas es intensivo. Con $n=m/M$ moles y $N=nN_A$ partículas
($N_A=\SI{6.022e23}{\per\mole}$) se definen las magnitudes \emph{específicas} y \emph{molares}:
\begin{equation}
x\equiv\frac{X}{m}\quad(\text{p.\,ej. } v=V/m),\qquad
\bar X\equiv\frac{X}{n}\quad(\text{p.\,ej. } \Vm=V/n).
\label{eq:molares}
\end{equation}

\begin{definicion}[Sistema simple; postulado de estado]
Un \emph{sistema simple} es homogéneo, isótropo, de composición fija, sin campos externos relevantes
y con un único modo de trabajo. Su estado de equilibrio queda fijado por \emph{dos} variables
independientes. En un sistema hidrostático, $p$, $V$ y $T$ están ligadas por $f(p,V,T)=0$.
\end{definicion}

\section{Equilibrio y procesos}
% =====================================================================

\subsection{Equilibrio termodinámico}

\begin{definicion}[Equilibrio termodinámico]
Un sistema está en \emph{equilibrio} cuando sus variables macroscópicas no cambian con el tiempo,
no hay flujos macroscópicos internos y las ligaduras no permiten ninguna evolución espontánea
adicional. Requiere, simultáneamente y en la medida en que las paredes lo permitan, equilibrio
\emph{mecánico} (sin fuerzas ni gradientes de $p$ no compensados), \emph{térmico} ($T$ uniforme a
través de paredes diatérmicas) y \emph{material} (sin difusión, reacción ni cambio de fase netos).
\end{definicion}

\begin{observacion}[Equilibrio frente a estado estacionario]
La independencia temporal no basta: una barra entre dos focos alcanza un perfil $T(x)$ constante
pero mantiene un flujo de calor, luego está en un \emph{estado estacionario}, no en equilibrio.
Se habla además de equilibrio \emph{interno} (cada subsistema), \emph{mutuo} (dos sistemas a través
de una pared permisiva) y \emph{ligado} (equilibrio interno, pero una ligadura impide el mutuo).
\end{observacion}

\subsection{Cambios de estado y procesos}

\begin{definicion}[Cambio de estado, proceso y función de estado]
Un \emph{cambio de estado} queda fijado por los estados inicial y final; un \emph{proceso} es el camino
que los une en el espacio de estados. Una \emph{función de estado} $Z=Z(X,Y)$ toma un valor único en
cada estado, de modo que $\Delta Z$ depende solo de los extremos y $\dd Z$ es una \emph{diferencial exacta}.
\end{definicion}

\begin{definicion}[Clasificación de procesos]
Un proceso es \emph{cuasiestático} si es lo bastante lento como para aproximarse por una sucesión de
estados de equilibrio (y por tanto representarse como una curva en el diagrama $p$--$V$);
\emph{reversible} si sistema \emph{y} alrededores pueden regresar a su estado inicial mediante cambios
infinitesimales; \emph{irreversible} en otro caso (gradientes finitos, fricción, expansión libre, \dots).
Se cumple: reversible $\subset$ cuasiestático; la lentitud no elimina la disipación.
\end{definicion}

\section{Ley cero y temperatura}
% =====================================================================

\begin{teorema}[Ley cero de la termodinámica]
Si dos sistemas $A$ y $B$ están, por separado, en equilibrio térmico con un tercero $C$, entonces $A$
y $B$ están en equilibrio térmico entre sí: $A\sim C$ y $B\sim C\Rightarrow A\sim B$.
\end{teorema}

Como todo sistema está en equilibrio consigo mismo (reflexividad) y la relación es simétrica, la ley
cero aporta la \emph{transitividad}: el equilibrio térmico es una \emph{relación de equivalencia}, y
sus clases de equivalencia definen la temperatura.

\begin{proposicion}[Existencia de la temperatura empírica]
Sean tres sistemas simples $\Sigma_i$ de variables $(p_i,V_i)$. Si el equilibrio térmico $\Sigma_1\sim\Sigma_3$
se expresa por $F_{13}(p_1,V_1,p_3,V_3)=0$ y $\Sigma_2\sim\Sigma_3$ por $F_{23}(p_2,V_2,p_3,V_3)=0$, existen
funciones $\theta_i(p_i,V_i)$ tales que el equilibrio térmico mutuo equivale a
\begin{equation}
\theta_1(p_1,V_1)=\theta_2(p_2,V_2)=\theta_3(p_3,V_3).
\label{eq:temp-empirica}
\end{equation}
\end{proposicion}

\begin{demo}
Despejando $p_3$ de ambas condiciones, $g_{13}(p_1,V_1,V_3)=g_{23}(p_2,V_2,V_3)$. Por la ley cero esta
igualdad equivale a $F_{12}(p_1,V_1,p_2,V_2)=0$, que no contiene $V_3$; luego $V_3$ debe cancelarse, lo
que solo es posible si cada miembro se escribe como $\psi(V_3,\theta_i(p_i,V_i))$ con la misma $\psi$.
Entonces $\theta_1=\theta_2$, e igualando con $p_3=\psi(V_3,\theta)$ se obtiene $\theta_3(p_3,V_3)=\theta$.
\end{demo}

\begin{observacion}[Lectura operativa]
En la práctica, se reescribe cada condición de equilibrio aislando \emph{la misma} combinación de las
variables del sistema común: $\theta_1(p_1,V_1)=\theta_3(p_3,V_3)=\theta_2(p_2,V_2)$. La función $\theta$
está definida salvo una función monótona arbitraria; las curvas $\theta_i(p_i,V_i)=\cte$ son las
\emph{isotermas} del sistema $i$. Es el método M1 (problema~1.1).
\end{observacion}

\begin{definicion}[Temperatura absoluta]
La \emph{temperatura termodinámica} $T$, de unidad el kelvin (K, sin signo de grado), es la escala
absoluta. Se relaciona con la escala Celsius por $T/\mathrm{K}=t/{}^\circ\mathrm{C}+273{,}15$; un
intervalo de \SI{1}{K} equivale a uno de \SI{1}{\celsius}, de modo que $\Delta T=\Delta t$.
\end{definicion}

\subsection{Ecuación de estado}

\begin{definicion}[Ecuación de estado]
Una \emph{ecuación (térmica o empírica) de estado} es una relación $f(p,V,T,n)=0$ entre variables de
equilibrio. Con $n$ fijo impone una ligadura: solo dos de las tres variables $p,V,T$ son independientes,
y cada una puede expresarse como función de las otras dos.
\end{definicion}

El ejemplo canónico es el \emph{gas ideal},
\begin{equation}
pV=nRT\qquad\Longleftrightarrow\qquad p\Vm=RT,\qquad R=\SI{8.314}{\joule\per\mole\per\kelvin},
\label{eq:gas-ideal}
\end{equation}
válido a densidades bajas y lejos de la condensación. Otros modelos usuales, que reaparecen en los
seminarios, se recogen en la tabla~\ref{tab:modelos}.

\subsection{Coeficientes termomecánicos}

\begin{definicion}[Coeficientes de respuesta]
Para un sistema hidrostático cerrado se definen el coeficiente de \emph{dilatación cúbica}, el de
\emph{compresibilidad isotérmica} y el \emph{piezotérmico}:
\begin{equation}
\alpha\equiv\frac{1}{V}\pd{V}{T}{p},\qquad
\kT\equiv-\frac{1}{V}\pd{V}{p}{T},\qquad
\beta\equiv\frac{1}{p}\pd{p}{T}{V},
\label{eq:coeficientes}
\end{equation}
o, de forma equivalente, $\alpha=\pd{\ln V}{T}{p}$ y $\kT=-\pd{\ln V}{p}{T}$.
\end{definicion}

Son intensivos (por eso pueden calcularse con $V$ o con $\Vm$ indistintamente) y miden la
sensibilidad \emph{relativa} del sistema (tabla~\ref{tab:coef}). En general dependen del estado,
$\alpha=\alpha(T,p)$; tratarlos como constantes es solo una aproximación local.

\subsection{Herramientas de cálculo en varias variables}

Si $x,y,z$ están ligadas por $f(x,y,z)=0$, cada una es función de las otras dos y se cumplen:
\begin{equation}
\pd{x}{y}{z}=\frac{1}{\pd{y}{x}{z}}\quad\text{(recíproca)},\qquad
\pd{x}{y}{z}\pd{y}{z}{x}\pd{z}{x}{y}=-1\quad\text{(cíclica)}.
\label{eq:ciclica}
\end{equation}

\begin{proposicion}[Relación entre coeficientes]
Los tres coeficientes no son independientes:
\begin{equation}
\alpha=p\,\beta\,\kT .
\label{eq:alpha-beta-kappa}
\end{equation}
\end{proposicion}

\begin{demo}
Aplicando la cíclica a $(p,V,T)$, $\pd{V}{T}{p}\pd{T}{p}{V}\pd{p}{V}{T}=-1$, y sustituyendo
$\pd{V}{T}{p}=\alpha V$, $\pd{T}{p}{V}=1/(p\beta)$ y $\pd{p}{V}{T}=-1/(\kT V)$ se obtiene
$-\alpha/(p\beta\kT)=-1$.
\end{demo}

\begin{teorema}[Criterio de exactitud]
La forma $\delta z=M(x,y)\,\dd x+N(x,y)\,\dd y$ es la diferencial exacta de una función $z(x,y)$ si y solo si
\begin{equation}
\pd{M}{y}{x}=\pd{N}{x}{y}.
\label{eq:schwarz}
\end{equation}
En tal caso $z$ se obtiene integrando $M$ respecto de $x$, $z=\int M\,\dd x+g(y)$, y fijando $g(y)$
con la condición $\partial z/\partial y=N$.
\end{teorema}

\subsection{Ecuación diferencial de estado}

Diferenciando $V(T,p)$, $p(T,V)$ y $T(p,V)$ y usando~\eqref{eq:coeficientes}:
\begin{equation}
\boxed{\;\dd\ln V=\alpha\,\dd T-\kT\,\dd p\;}\qquad
\dd p=p\beta\,\dd T-\frac{\dd V}{\kT V},\qquad
\dd T=\frac{\dd V}{\alpha V}+\frac{\dd p}{p\beta}.
\label{eq:diferenciales}
\end{equation}
Conocidos dos coeficientes (el tercero sale de~\eqref{eq:alpha-beta-kappa}), la ecuación de estado se
reconstruye integrando cualquiera de ellas: es el método M5. Como $\ln V$ es función de estado, la
integral no depende del camino; por una trayectoria rectangular $(T_a,p_a)\to(T_b,p_a)\to(T_b,p_b)$,
\begin{equation}
\ln\frac{V_b}{V_a}=\int_{T_a}^{T_b}\alpha(T,p_a)\,\dd T-\int_{p_a}^{p_b}\kT(T_b,p)\,\dd p
\;\xrightarrow{\ \alpha,\kT\ \text{ctes}\ }\;
\alpha\,\Delta T-\kT\,\Delta p\approx\frac{\Delta V}{V}.
\label{eq:finitos}
\end{equation}

\begin{observacion}[Sólidos: dilatación lineal]
Para un sólido isótropo, $V\propto L^3$ y $\alpha=3\alpha_l$, con $\alpha_l\equiv\frac{1}{L}\pd{L}{T}{p}$.
Para $\alpha_l\Delta T\ll1$, $\Delta L\simeq\alpha_l L_0\Delta T$ (problemas 1.5 y 1.6). A volumen
constante, $\dd V=0$ en~\eqref{eq:diferenciales} da $\pd{p}{T}{V}=\alpha/\kT$ (problema 1.7).
\end{observacion}

\begin{ejemplo}[Coeficientes del gas de van der Waals]\label{ej:vdw}
De $\bigl(p+a/\Vm^2\bigr)(\Vm-b)=RT$ no es práctico despejar $\Vm$ (cúbica), pero sí $T$ y $p$. Por la
regla recíproca,
\begin{equation*}
\frac{1}{\Vm\alpha}=\pd{T}{\Vm}{p}=\frac{1}{R}\Bigl(p-\frac{a}{\Vm^2}+\frac{2ab}{\Vm^3}\Bigr),\qquad
-\frac{1}{\Vm\kT}=\pd{p}{\Vm}{T}=-\frac{RT}{(\Vm-b)^2}+\frac{2a}{\Vm^3},
\end{equation*}
y eliminando $p$ con la ecuación de estado,
\begin{equation*}
\alpha=\frac{R\Vm^2(\Vm-b)}{RT\Vm^3-2a(\Vm-b)^2},\qquad
\kT=\frac{\Vm^2(\Vm-b)^2}{RT\Vm^3-2a(\Vm-b)^2}.
\end{equation*}
Con $a=b=0$ se recuperan los del gas ideal, $\alpha=1/T$ y $\kT=1/p$.
\end{ejemplo}

\begin{tcolorbox}[apunte,title={\color{acento}\espaciado{Los cinco métodos}},
  coltitle=acento,colbacktitle=fondo,attach title to upper,after title={\par\smallskip}]
\begin{description}[leftmargin=2.2em,style=sameline,font=\normalfont\bfseries\color{marino},itemsep=3pt]
\item[M1.] \emph{Ley cero.} Reescribir cada condición de equilibrio aislando la misma combinación
$\theta_3(p_3,V_3)$ del sistema común; lo que queda a cada lado son $\theta_1$ y $\theta_2$, y
$\Sigma_1\sim\Sigma_2\iff\theta_1=\theta_2$.
\item[M2.] \emph{Politrópico.} $p_1V_1^n=p_2V_2^n\Rightarrow p_2=p_1(V_1/V_2)^n$ y
$n=\ln(p_2/p_1)/\ln(V_1/V_2)$. Unificar unidades \emph{antes} de operar.
\item[M3.] \emph{Coeficientes constantes.} $\Delta L\simeq\alpha_lL_0\Delta T$; $\ln(V_2/V_1)=\alpha\Delta T-\kT\Delta p$;
a volumen fijo, $\Delta p=(\alpha/\kT)\Delta T$.
\item[M4.] \emph{Derivación.} Despejar la variable que se deriva si es sencillo; si no, usar la regla
recíproca~\eqref{eq:ciclica} o derivar implícitamente. Expresar el resultado en las variables pedidas.
\item[M5.] \emph{Integración.} (1) Completar el par necesario con $\alpha=p\beta\kT$; (2) escribir la
diferencial adecuada~\eqref{eq:diferenciales}; (3) comprobar exactitud~\eqref{eq:schwarz}; (4) integrar
respecto de una variable y fijar la función de la otra; (5) fijar la constante con un límite físico
(normalmente gas ideal para $\Vm\to\infty$).
\end{description}
\end{tcolorbox}

\begin{observacion}[Elección de la diferencial en M5]
Si los datos son $\alpha$ y $\kT$, conviene $\dd\ln V$; si son $\beta$ y $\kT$ y se pide $p(T,V)$, conviene
$\dd p$ o $\dd\ln p$. Cuando los coeficientes contienen $\Vm-A$ o $A+aT-bp$, conviene buscar la diferencial
logarítmica de esa combinación: la integral es inmediata.
\end{observacion}

\subsection{Problema 1.1. Temperaturas empíricas de tres gases}

\begin{problema}{1.1}{Ley cero con tres gases}
Los sistemas $\Sigma_1$, $\Sigma_2$ y $\Sigma_3$ son gases de variables $(p_i,V_i)$. Cuando $\Sigma_1$ y $\Sigma_3$
están en equilibrio térmico se cumple $p_1V_1-bp_1-p_3V_3=0$, y cuando lo están $\Sigma_2$ y $\Sigma_3$,
$p_2V_2-p_3V_3+b'p_3V_3/V_2=0$, con $b,b'$ constantes. ¿Qué relación expresa el equilibrio térmico entre
$\Sigma_1$ y $\Sigma_2$? ¿Cuáles son las funciones $\theta_i(p_i,V_i)$ que coinciden en el equilibrio?
\end{problema}

\resolucion\idea Método M1: aislar en ambas condiciones la misma combinación de $\Sigma_3$, que aparece
de forma natural como $p_3V_3$.

\paso{Paso 1: condición $\Sigma_1\sim\Sigma_3$} $p_1(V_1-b)=p_3V_3$.

\paso{Paso 2: condición $\Sigma_2\sim\Sigma_3$} Sacando factor común $p_3V_3$:
$p_3V_3\,(1-b'/V_2)=p_2V_2$, es decir, $p_3V_3=\dfrac{p_2V_2^2}{V_2-b'}$.

\paso{Paso 3: ley cero} Igualando ambas expresiones de $p_3V_3$:
\begin{equation*}
\resultado{p_1(V_1-b)=\frac{p_2V_2^2}{V_2-b'}}\qquad
\resultado{\theta_1=p_1(V_1-b),\quad \theta_2=\frac{p_2V_2^2}{V_2-b'},\quad \theta_3=p_3V_3}
\end{equation*}
\fin

\begin{observacion}[Unicidad]
Cualquier función monótona $\phi(\theta)$ aplicada a las tres $\theta_i$ es igualmente válida; la elección
anterior es la más simple. $\Sigma_3$ es un gas ideal ($\theta_3\propto T$), $\Sigma_1$ un gas de Clausius y
$\Sigma_2$ un gas con corrección en $b'/V_2$.
\end{observacion}

\subsection{Problema 1.2. Ecuación de estado a partir de \texorpdfstring{$\dd V$}{dV}}

\begin{problema}{1.2}{Integración directa}
Un fluido pasa de $(T,p)$ a $(T+\dd T,p+\dd p)$ con $\dd V=\dfrac{2V}{T}\,\dd T-\dfrac{3V}{p}\,\dd p$.
Encuéntrese la relación entre $V$, $p$ y $T$.
\end{problema}

\resolucion\idea Dividir por $V$: la diferencial queda en variables separadas (método M5 con
$\alpha=2/T$ y $\kT=3/p$).

\paso{Paso 1: diferencial logarítmica} $\dd\ln V=2\,\dd\ln T-3\,\dd\ln p=\dd\ln\bigl(T^2p^{-3}\bigr)$, exacta de forma trivial.

\paso{Paso 2: integración} $\ln V=\ln(T^2/p^3)+\ln K$, de donde
\begin{equation*}
\resultado{p^3V=K\,T^2}
\end{equation*}
\fin

\subsection{Problema 1.3. Proceso politrópico: presión final}

\begin{problema}{1.3}{Expansión con $pV^{7/5}=\cte$}
Un gas efectúa un proceso reversible con $pV^{7/5}=\cte$ desde $p_1=\SI{1}{bar}$, $V_1=\SI{1}{m^3}$
hasta $V_2=\SI{3e6}{cm^3}$. Calcúlese $p_2$ en kPa y represéntese el proceso en el diagrama $pV$.
\end{problema}

\resolucion\idea Método M2 tras unificar unidades: $V_2=\SI{3e6}{cm^3}=\SI{3}{m^3}$, $p_1=\SI{100}{kPa}$.

\paso{Paso 1: presión final}
\begin{equation*}
p_2=p_1\Bigl(\frac{V_1}{V_2}\Bigr)^{7/5}=\SI{100}{kPa}\cdot3^{-1{,}4}=\frac{\SI{100}{kPa}}{4{,}656}
\quad\Longrightarrow\quad\resultado{p_2\simeq\SI{21.5}{kPa}}
\end{equation*}

\paso{Paso 2: representación} El proceso es una expansión cuasiestática que, en el plano $pV$, cae más
deprisa que la isoterma $pV=\cte$ que pasa por el estado inicial (figura~\ref{fig:p13}).
\fin

\subsection{Problema 1.5. Reloj de péndulo}

\begin{problema}{1.5}{Péndulo de cobre}
Un reloj con un péndulo de cobre de \SI{0.5}{m}, calibrado a \SI{20}{\celsius}, funciona un día a
\SI{40}{\celsius}. ¿Se adelanta o se atrasa? ¿Cuánto? ($\alpha_l=\SI{17e-6}{\per\kelvin}$.)
\end{problema}

\resolucion\idea El periodo $\mathcal T=2\pi\sqrt{l/g}$ crece con la longitud. El reloj cuenta
oscilaciones: si cada una dura más, marca menos tiempo del real y se \emph{atrasa}. Método M3.

\paso{Paso 1: variación relativa del periodo} Con $l=l_0(1+\alpha_l\Delta T)$ y $\Delta T=\SI{20}{K}$,
\begin{equation*}
\frac{\mathcal T}{\mathcal T_0}=\sqrt{1+\alpha_l\Delta T}\simeq1+\tfrac12\alpha_l\Delta T
\quad\Longrightarrow\quad
\frac{\Delta\mathcal T}{\mathcal T_0}\simeq\tfrac12\alpha_l\Delta T=\num{1.70e-4}.
\end{equation*}

\paso{Paso 2: retraso diario} En un tiempo real $t$ el reloj marca $t\,\mathcal T_0/\mathcal T$; el retraso
es $t\,(1-\mathcal T_0/\mathcal T)\simeq t\,\Delta\mathcal T/\mathcal T_0$. Para $t=\SI{86400}{s}$:
\begin{equation*}
\resultado{\text{se atrasa}\ \simeq\SI{14.7}{s/día}}
\end{equation*}
\fin

\begin{observacion}[La longitud no interviene]
El resultado no depende de $l_0=\SI{0.5}{m}$ ni de $g$: solo importa la variación \emph{relativa},
$\tfrac12\alpha_l\Delta T$. Es el rasgo general de los coeficientes de respuesta.
\end{observacion}

\subsection{Problema 1.6. Dilatación de un puente}

\begin{problema}{1.6}{Puente de acero}
Un puente de acero continuo mide \SI{100}{m}. ¿Cuánto varía su longitud entre \SI{-10}{\celsius} y
\SI{50}{\celsius}? ($\alpha_l=\SI{11e-6}{\per\kelvin}$.)
\end{problema}

\resolucion\idea Método M3 con $\alpha_l\Delta T\ll1$ y $\Delta T=\Delta t=\SI{60}{K}$.
\begin{equation*}
\Delta L\simeq\alpha_l L_0\,\Delta T=\SI{11e-6}{\per\kelvin}\cdot\SI{100}{m}\cdot\SI{60}{K}=\SI{0.066}{m}
\quad\Longrightarrow\quad\resultado{\Delta L\simeq\SI{6.6}{cm}}
\end{equation*}
\fin

\subsection{Problema 1.7. Mercurio calentado a volumen constante}

\begin{problema}{1.7}{Presión a volumen fijo}
Calcúlese el aumento de presión de una masa de mercurio calentada a volumen constante de \SI{0}{\celsius}
a \SI{1}{\celsius}, con $\alpha=\SI{18e-5}{\per\kelvin}$ y $\kT=\SI{39e-12}{\per\pascal}$ constantes.
\end{problema}

\resolucion\idea A volumen fijo, $\dd\ln V=0$ en~\eqref{eq:diferenciales}: $\alpha\,\dd T=\kT\,\dd p$.

\paso{Paso 1: integración}
\begin{equation*}
\Delta p=\frac{\alpha}{\kT}\,\Delta T=\frac{\SI{18e-5}{\per\kelvin}}{\SI{39e-12}{\per\pascal}}\cdot\SI{1}{K}=\SI{4.615e6}{Pa}
\quad\Longrightarrow\quad\resultado{\Delta p\simeq\SI{46.2}{bar}}
\end{equation*}
\fin

\begin{observacion}[Orden de magnitud]
Un solo kelvin produce unas 46 atmósferas: los líquidos son casi incompresibles ($\kT$ minúsculo), por
lo que impedir su dilatación exige presiones enormes. De aquí el riesgo de un termómetro sobrecalentado.
Nótese que $\alpha/\kT=p\beta=\pd{p}{T}{V}$.
\end{observacion}

\subsection{Problema 1.8. Coeficientes de la ecuación de Clausius}

\begin{problema}{1.8}{Ecuación de Clausius}
Para la ecuación de estado $p(\Vm-b)=RT$, calcúlense $\alpha$ y $\kT$.
\end{problema}

\resolucion\idea Método M4: aquí $\Vm$ se despeja sin dificultad, $\Vm=RT/p+b$.

\paso{Paso 1: derivadas}
$\pd{\Vm}{T}{p}=\dfrac{R}{p}$, \quad $\pd{\Vm}{p}{T}=-\dfrac{RT}{p^2}$.

\paso{Paso 2: coeficientes} Dividiendo por $\Vm=(RT+bp)/p$:
\begin{equation*}
\resultado{\alpha=\frac{R}{RT+bp}}\qquad\resultado{\kT=\frac{RT}{p\,(RT+bp)}}
\end{equation*}

\paso{Comprobación} $\beta=\frac1p\pd{p}{T}{\Vm}=\frac{1}{p}\frac{R}{\Vm-b}=\frac1T$, y en efecto
$p\beta\kT=\frac{R}{RT+bp}=\alpha$. Con $b=0$, $\alpha=1/T$ y $\kT=1/p$ (gas ideal).
\fin

\subsection{Problema 1.9. Ecuación de estado a partir de \texorpdfstring{$\alpha$}{alpha} y \texorpdfstring{$\kappa_T$}{kT} (I)}

\begin{problema}{1.9}{$\alpha=3/T$, $\kT=2/p$}
Determínese la ecuación de estado de un sistema hidrostático con $\alpha=3/T$ y $\kT=2/p$.
\end{problema}

\resolucion\idea Método M5 con $\dd\ln\Vm$; las variables están separadas.
\begin{equation*}
\dd\ln\Vm=\frac{3\,\dd T}{T}-\frac{2\,\dd p}{p}=\dd\ln\frac{T^3}{p^2}
\quad\Longrightarrow\quad\resultado{p^2\Vm=\cte\cdot T^3}
\end{equation*}
\fin

% ---------------------------------------------------------------------
\subsection{Problema 1.10. Ecuación de estado a partir de \texorpdfstring{$\alpha$}{alpha} y \texorpdfstring{$\kappa_T$}{kT} (II)}

\begin{problema}{1.10}{Gas con corrección lineal en $p$}
Determínese la ecuación de estado de un gas con $\alpha=\dfrac{R}{p\Vm}$ y $\kT=\dfrac{RT}{p^2\Vm}+\dfrac{B}{2\Vm}$.
\end{problema}

\resolucion\idea Método M5. Multiplicando por $\Vm$ desaparece el volumen de los coeficientes y se
integra $\dd\Vm$ directamente.

\paso{Paso 1: diferencial}
$\dd\Vm=\Vm\alpha\,\dd T-\Vm\kT\,\dd p=\dfrac{R}{p}\,\dd T-\Bigl(\dfrac{RT}{p^2}+\dfrac B2\Bigr)\dd p$.

\paso{Paso 2: exactitud} $\partial_p(R/p)=-R/p^2=\partial_T\bigl(-RT/p^2-B/2\bigr)$. \checkmark

\paso{Paso 3: integración} $\Vm=\int\frac Rp\,\dd T=\frac{RT}{p}+g(p)$; imponiendo
$\partial_p\Vm=-\frac{RT}{p^2}+g'(p)=-\frac{RT}{p^2}-\frac B2$ resulta $g=-\frac{B}{2}p+\Vm_0$.

\paso{Paso 4: constante} Para $B\to0$ los coeficientes son los del gas ideal y debe recuperarse
$p\Vm=RT$, luego $\Vm_0=0$:
\begin{equation*}
\resultado{p\Vm=RT-\tfrac12Bp^2}
\end{equation*}
\fin

% ---------------------------------------------------------------------
\subsection{Problema 1.11. Ecuación de estado a partir de \texorpdfstring{$\beta$}{beta} y \texorpdfstring{$\kappa_T$}{kT}}

\begin{problema}{1.11}{$\beta=aT/(bp^2)$, $\kT=bp/\Vm$}
Determínese la ecuación de estado de un sistema hidrostático con $\beta=\dfrac{aT}{bp^2}$ y
$\kT=\dfrac{bp}{\Vm}$, siendo $a$ y $b$ constantes.
\end{problema}

\resolucion\idea Método M5: completar el par con $\alpha=p\beta\kT$ y usar $\dd\Vm$.

\paso{Paso 1: tercer coeficiente} $\alpha=p\cdot\dfrac{aT}{bp^2}\cdot\dfrac{bp}{\Vm}=\dfrac{aT}{\Vm}$.

\paso{Paso 2: diferencial e integración} $\dd\Vm=\Vm\alpha\,\dd T-\Vm\kT\,\dd p=aT\,\dd T-bp\,\dd p$, exacta
y de variables separadas:
\begin{equation*}
\resultado{\Vm=\frac{aT^2}{2}-\frac{bp^2}{2}+\cte}
\end{equation*}
\fin

\subsection{Problema 1.12. Ecuación de estado a partir de \texorpdfstring{$\alpha$}{alpha} y \texorpdfstring{$\beta$}{beta}}

\begin{problema}{1.12}{$\alpha$ y $\beta$ con denominador común}
Determínese la ecuación de estado de un fluido con $\alpha=\dfrac{a}{A+aT-bp}$ y $\beta=\dfrac{a}{bp}$,
siendo $A$, $a$ y $b$ constantes.
\end{problema}

\resolucion\idea Método M5. Tras hallar $\kT$, la diferencial es la logarítmica de $A+aT-bp$.

\paso{Paso 1: tercer coeficiente} $\kT=\dfrac{\alpha}{p\beta}=\dfrac{a}{A+aT-bp}\cdot\dfrac{bp}{ap}=\dfrac{b}{A+aT-bp}$.

\paso{Paso 2: integración}
\begin{equation*}
\dd\ln\Vm=\frac{a\,\dd T-b\,\dd p}{A+aT-bp}=\dd\ln(A+aT-bp)
\quad\Longrightarrow\quad\resultado{\Vm=\cte\cdot(A+aT-bp)}
\end{equation*}
\fin

\subsection{Problema 1.13. ¿Se recupera el gas ideal?}

\begin{problema}{1.13}{Coeficientes con $\Vm-A$}
Un sistema hidrostático tiene $\alpha=\dfrac{\Vm-A}{2\Vm T}$ y $\kT=\dfrac{\Vm-A}{p\Vm}$ ($A$ constante).
Determínese la ecuación térmica de estado. ¿Puede recuperarse el comportamiento de gas ideal?
\end{problema}

\resolucion\idea Método M5. Al multiplicar por $\Vm$ aparece $\Vm-A$ en ambos términos: se integra
$\dd\ln(\Vm-A)$.

\paso{Paso 1: diferencial}
$\dd\Vm=\dfrac{\Vm-A}{2T}\,\dd T-\dfrac{\Vm-A}{p}\,\dd p
\;\Longrightarrow\;
\dd\ln(\Vm-A)=\dfrac{\dd T}{2T}-\dfrac{\dd p}{p}$.

\paso{Paso 2: integración}
\begin{equation*}
\resultado{p\,(\Vm-A)=K\sqrt T}
\end{equation*}

\paso{Paso 3: límite} Para $\Vm\gg A$, $p\Vm\to K\sqrt T$ y $\alpha\to1/(2T)\neq1/T$: \emph{no} se recupera
el gas ideal en ningún límite, porque la dependencia en $T$ es $T^{1/2}$ y no $T$.
\fin

\begin{observacion}[Lectura del enunciado]
Si el coeficiente fuera $\alpha=(\Vm-A)/(\Vm T)$, el mismo procedimiento daría $p(\Vm-A)=RT$, la
ecuación de Clausius, que sí tiende al gas ideal para $\Vm\gg A$. Conviene comprobar siempre el límite:
$\alpha\to1/T$ y $\kT\to1/p$.
\end{observacion}

\subsection{Problema 1.14. Ecuación de Dieterici}

\begin{problema}{1.14}{$p(T,\Vm)$ a partir de $\kT$ y $\beta$}
Un gas tiene $\kT=\dfrac{RT\Vm(\Vm-b)}{p\,[RT\Vm^2-a(\Vm-b)]}$ y $\beta=\dfrac{RT\Vm+a}{RT^2\Vm}$.
Hállese $p(T,\Vm)$ sabiendo que para $\Vm$ grande el gas se comporta como ideal.
\end{problema}

\resolucion\idea Método M5 con $\dd\ln p=\beta\,\dd T-\dfrac{\dd\Vm}{p\kT\Vm}$: los dos coeficientes ya son
los de las variables pedidas.

\paso{Paso 1: diferencial} Simplificando cada término,
\begin{equation*}
\dd\ln p=\Bigl(\frac1T+\frac{a}{RT^2\Vm}\Bigr)\dd T-\Bigl(\frac{1}{\Vm-b}-\frac{a}{RT\Vm^2}\Bigr)\dd\Vm .
\end{equation*}

\paso{Paso 2: exactitud} $\partial_{\Vm}\bigl(\tfrac1T+\tfrac{a}{RT^2\Vm}\bigr)=-\tfrac{a}{RT^2\Vm^2}
=\partial_T\bigl(-\tfrac{1}{\Vm-b}+\tfrac{a}{RT\Vm^2}\bigr)$. \checkmark

\paso{Paso 3: integración} Respecto de $T$: $\ln p=\ln T-\dfrac{a}{RT\Vm}+g(\Vm)$. Derivando respecto de
$\Vm$ e igualando, $g'(\Vm)=-1/(\Vm-b)$, luego $g=-\ln(\Vm-b)+\ln K$.

\paso{Paso 4: constante} Para $\Vm\to\infty$, $p\to KT/\Vm$; el límite ideal exige $K=R$:
\begin{equation*}
\resultado{p=\frac{RT}{\Vm-b}\,\exp\Bigl(-\frac{a}{RT\Vm}\Bigr)}
\end{equation*}
\fin

\subsection{Problema 1.15. Coeficiente de dilatación de un gas del virial}

\begin{problema}{1.15}{$\alpha$ en un estado dado}
Un gas tiene $\kT=\dfrac{1}{RT\,(1/\Vm+2b/\Vm^2)}$ y $\beta=\dfrac1T$, con $b=\SI{4e-2}{L/mol}$.
Calcúlese $\alpha$ cuando un mol ocupa \SI{1}{L} a \SI{300}{K}, sabiendo que para $\Vm$ grande el gas es ideal.
\end{problema}

\resolucion\idea $\alpha=p\beta\kT$ exige conocer $p$ en ese estado, luego antes hay que reconstruir
$p(T,\Vm)$ (método M5). Como $\kT$ contiene $p$ implícitamente, se trabaja con $\dd p$.

\paso{Paso 1: diferencial}
$\dd p=p\beta\,\dd T-\dfrac{\dd\Vm}{\kT\Vm}=\dfrac pT\,\dd T-RT\Bigl(\dfrac1{\Vm^2}+\dfrac{2b}{\Vm^3}\Bigr)\dd\Vm$.

\paso{Paso 2: integración} Respecto de $\Vm$: $p=RT\bigl(\tfrac1{\Vm}+\tfrac{b}{\Vm^2}\bigr)+f(T)$. Imponiendo
$\partial_Tp=p/T$ se obtiene $f'=f/T$, es decir, $f=cT$. El límite ideal exige $c=0$:
\begin{equation*}
p=\frac{RT}{\Vm}\Bigl(1+\frac{b}{\Vm}\Bigr).
\end{equation*}

\paso{Paso 3: coeficiente de dilatación}
\begin{equation*}
\alpha=p\beta\kT=\frac{1/\Vm+b/\Vm^2}{T\,(1/\Vm+2b/\Vm^2)}=\frac{\Vm+b}{T\,(\Vm+2b)}
=\frac{1{,}04}{300\cdot1{,}08}\,\si{\per\kelvin}
\quad\Longrightarrow\quad\resultado{\alpha\simeq\SI{3.21e-3}{\per\kelvin}}
\end{equation*}
\fin

\begin{observacion}[Comparación con el gas ideal]
El gas ideal daría $\alpha=1/T=\SI{3.33e-3}{\per\kelvin}$; la corrección relativa es $-b/(\Vm+2b)\simeq\SI{-3.7}{\percent}$,
del orden de $b/\Vm$, como corresponde a un desarrollo del virial truncado.
\end{observacion}

\subsection{Problema 1.16. Cobre calentado y comprimido}

\begin{problema}{1.16}{Volumen final con coeficientes medios}
Un kilogramo de cobre se calienta de \SI{300}{K} a \SI{500}{K} mientras su presión aumenta \SI{1000}{bar}.
Con $\alpha=\SI{51.7e-6}{\per\kelvin}$, $\kT=\SI{0.8065e-11}{\per\pascal}$ (valores medios) y
$v_0=\SI{111.5e-6}{m^3/kg}$, calcúlese el volumen final.
\end{problema}

\resolucion\idea Método M3: integrar $\dd\ln V=\alpha\,\dd T-\kT\,\dd p$ con coeficientes constantes.
Como $\ln V$ es función de estado, no importa el orden en que se calienta y se comprime.

\paso{Paso 1: integración}
\begin{align*}
\ln\frac{V}{V_0}&=\alpha\Delta T-\kT\Delta p=\SI{51.7e-6}{\per\kelvin}\cdot\SI{200}{K}-\SI{0.8065e-11}{\per\pascal}\cdot\SI{1e8}{Pa}\\
&=\num{1.034e-2}-\num{8.065e-4}=\num{9.534e-3}.
\end{align*}

\paso{Paso 2: volumen final} Con $V_0=mv_0=\SI{111.5e-6}{m^3}$,
\begin{equation*}
V=V_0\,e^{\num{9.534e-3}}\quad\Longrightarrow\quad\resultado{V\simeq\SI{112.57e-6}{m^3}}
\end{equation*}
\fin

\begin{observacion}[Competencia de efectos]
La dilatación térmica (\SI{+1.03}{\percent}) domina sobre la compresión (\SI{-0.08}{\percent}). Como
$\ln(V/V_0)\ll1$, la aproximación lineal $\Delta V\simeq V_0(\alpha\Delta T-\kT\Delta p)$ da el mismo
resultado con tres cifras.
\end{observacion}

\subsection{Parcial 2025/26, cuestión 1}

\begin{problema}{P.1}{Parcial 2025/26}
Una sustancia hipotética tiene $\alpha=\dfrac{1}{T+A}$ y $\kT=\dfrac1p$. (a) Calcule la ecuación de estado
$p=p(T,V)$. (b) Compruebe explícitamente la relación cíclica entre $p$, $V$ y $T$. (c) En una expansión
espontánea de este gas (presión inicial \SI{1}{bar}) hasta el equilibrio con unos alrededores a
$p_{\mathrm{ext}}=\SI{2}{bar}$, ¿pueden usarse estos coeficientes para describir la curva del proceso?
\end{problema}

\resolucion\idea (a) Método M5; (b) método M4 sobre el resultado; (c) concepto de proceso cuasiestático.

\paso{(a) Ecuación de estado}
$\dd\ln V=\dfrac{\dd T}{T+A}-\dfrac{\dd p}{p}=\dd\ln\dfrac{T+A}{p}$, luego $pV=K(T+A)$ y
\begin{equation*}
\resultado{p=\frac{K\,(T+A)}{V}}
\end{equation*}

\paso{(b) Relación cíclica} De $pV=K(T+A)$:
\begin{equation*}
\pd{p}{V}{T}=-\frac{K(T+A)}{V^2}=-\frac pV,\qquad
\pd{V}{T}{p}=\frac Kp,\qquad
\pd{T}{p}{V}=\frac VK,
\end{equation*}
y su producto es $\bigl(-\tfrac pV\bigr)\tfrac Kp\,\tfrac VK=-1$.~\checkmark\par
Equivalentemente, $\beta=\frac{1}{p}\frac{K}{V}=\frac{1}{T+A}$ y $p\beta\kT=\frac{1}{T+A}=\alpha$.

\paso{(c) Proceso espontáneo} \textbf{No.} Una evolución espontánea con un desequilibrio finito de presión
es irreversible y no cuasiestática: los estados intermedios no son de equilibrio, $p$ y $T$ no están
definidas uniformemente y el proceso no es una curva del espacio de estados. Los coeficientes
(y la ecuación de estado) solo relacionan los estados \emph{inicial y final}, ambos de equilibrio.
\fin

\begin{observacion}[Detalle del enunciado]
Con $p_{\mathrm{ext}}>p_i$ el proceso espontáneo sería en realidad una \emph{compresión}; la respuesta no
cambia, porque depende solo de que el proceso sea no cuasiestático. Tampoco la opción «solo si es
adiabática» la salva: ser adiabático no lo hace cuasiestático.
\end{observacion}
